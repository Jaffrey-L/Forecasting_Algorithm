def test_calculate_wmape_statistics_basic(sample_forecast_data):

    assert 'std' in stats['overall']
    assert 'min' in stats['overall']
    assert 'max' in stats['overall']

    assert 'threshold_exceeded_rate' in stats['overall']

def test_calculate_wmape_statistics_threshold(sample_forecast_data):

#### test_calculate_wmape_statistics_grouping

**测试目标**: 验证分组统计计算

```python
def test_calculate_wmape_statistics_grouping(sample_forecast_data):
    # 准备测试数据
    df = sample_forecast_data
    
    # 调用方法
    from src.forecasting.monitor import ForecastingMonitor
    monitor = ForecastingMonitor('test_db', wmape_threshold=0.30)
    stats = monitor.calculate_wmape_statistics(df)
    
    # 验证SPU分组统计
    for spu in df['spu'].unique():
        spu_data = df[df['spu'] == spu]
        assert spu in stats['by_spu']
        assert stats['by_spu'][spu]['count'] == len(spu_data)
        assert stats['by_spu'][spu]['mean_wmape'] == spu_data['validation_wmape'].mean()
    
    # 验证日期分组统计
    for date in df['run_date'].unique():
        date_data = df[df['run_date'] == date]
        date_str = str(date.date())
        assert date_str in stats['by_date']
        assert stats['by_date'][date_str]['count'] == len(date_data)
    
    # 验证模型分组统计
    for model in df['winner_algo'].unique():
        model_data = df[df['winner_algo'] == model]
        assert model in stats['by_model']
        assert stats['by_model'][model]['count'] == len(model_data)
```

```python
def test_detect_anomalies_no_duplicate(sample_forecast_data):
    # 准备测试数据
    df = sample_forecast_data.copy()
    
    # 确保有超阈值的数据
    df.loc[0, 'validation_wmape'] = 0.45
    
    # 调用方法
    from src.forecasting.monitor import ForecastingMonitor
    monitor = ForecastingMonitor('test_db', wmape_threshold=0.30)
    anomalies = monitor.detect_anomalies(df)
    
    # 验证无重复
    for anomaly in anomalies:
        if anomaly['type'] == 'WMAPE_THRESHOLD_EXCEEDED':
            # 超阈值的不应同时是异常高
            high_anomalies = [a for a in anomalies if a['type'] == 'WMAPE_HIGH']
            for high in high_anomalies:
                assert not (high['spu'] == anomaly['spu'] and high['run_date'] == anomaly['run_date'])
```
